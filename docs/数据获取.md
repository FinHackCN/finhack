# 数据获取Collector系统详细分析

## 整体架构概述

Finhack框架的数据获取系统采用分层架构设计，通过Tushare API实现金融数据的自动化收集与管理。整个系统以**TushareCollector**为核心控制器，协调各个专用数据收集模块，实现多线程并发数据获取。

### 3. 多线程管理机制

#### 3.1 线程创建与管理

```python
def mTread(self, className, functionName, dependency_check=None):
    """创建并启动数据采集线程"""
    # 1. 依赖关系检查
    if dependency_check and not self.check_dependency(dependency_check):
        Log.logger.warning(f"跳过线程 {functionName}，因为依赖 {dependency_check} 未满足")
        return
    
    # 2. 并发控制
    max_concurrent_threads = 20  # 最大并发线程数
    active_threads = sum(1 for t in self.thread_list if t.is_alive())
    
    if active_threads >= max_concurrent_threads:
        # 等待部分线程完成
        while active_threads >= max_concurrent_threads:
            completed_threads = [t for t in self.thread_list if not t.is_alive()]
            for t in completed_threads:
                if t in self.thread_list:
                    self.thread_list.remove(t)
            active_threads = sum(1 for t in self.thread_list if t.is_alive())
            if active_threads >= max_concurrent_threads:
                time.sleep(1)
    
    # 3. 创建线程
    thread = collectThread(className, functionName, self.pro, self.db)
    thread.set_max_runtime(self.max_thread_runtime)
    self.thread_list.append(thread)
    time.sleep(0.1)  # 防止同时启动过多线程
```

#### 3.2 collectThread线程类

```python
class collectThread(threading.Thread):
    def __init__(self, className, functionName, pro, db):
        threading.Thread.__init__(self)
        self.className = className
        self.functionName = functionName
        self.pro = pro
        self.db = db
        self.max_runtime = None
        
    def run(self):
        # 1. 超时监控
        start_time = time.time()
        if self.max_runtime:
            # 创建守护线程监控超时
            pass
        
        # 2. 执行具体的数据收集方法
        method = getattr(self.className, self.functionName)
        method(pro=self.pro, db=self.db)
        
        # 3. 记录执行时间
        end_time = time.time()
        Log.logger.info(f"线程 {self.functionName} 完成，运行时间: {(end_time-start_time)/60:.2f} 分钟")
```

def check_dependency(self, table_name):
    """检查依赖表是否存在，如果不存在但是有创建该表的功能则尝试创建"""
    if table_name in self.dependency_status:
        return self.dependency_status[table_name]
    else:
        # 检查表是否存在
        adapter = DB.get_adapter(self.db)
        if adapter.table_exists(table_name):
            result = DB.selectToList(f"SELECT 1 FROM {table_name} LIMIT 1", self.db)
            return len(result) > 0
        else:
            # 表不存在，尝试自动创建
            if table_name == 'astock_basic':
                success = tsAStockBasic.stock_basic(self.pro, self.db)
                if success:
                    self.dependency_status['astock_basic'] = True
                    return True
            # ... 其他表的自动创建逻辑

def save_checkpoint(cls, api, table, last_date, last_ts_code=None):
    """保存检查点信息"""

    data = {
        "last_date": last_date,
        "last_ts_code": last_ts_code,
        "update_time": datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    }
    with open(checkpoint_path, "w") as f:
        json.dump(data, f)

@classmethod
def load_checkpoint(cls, api, table):
    """加载检查点信息"""
    checkpoint_path = cls.get_checkpoint_path(api, table)
    if not os.path.exists(checkpoint_path):
        return None
    with open(checkpoint_path, "r") as f:
        data = json.load(f)
    return data

**增量获取逻辑**：
```python
def getPrice(pro, api, table, db):
    # 1. 检查点恢复
    checkpoint = tsAStockPrice.load_checkpoint(api, table)
    if checkpoint and checkpoint.get("last_date"):
        lastdate = checkpoint["last_date"]
        # 删除检查点日期的数据，重新获取
        DB.delete(f"DELETE FROM {table} WHERE trade_date = '{lastdate}'", db)
    else:
        # 从数据库获取最后日期
        lastdate = tsSHelper.getLastDateAndDelete(table, 'trade_date', "", db)
    
    # 2. 按批次获取数据
    start_date = datetime.datetime.strptime(lastdate, "%Y%m%d").date()
    end_date = datetime.datetime.now().date()
    batch_days = 1  # 每批次1天
    
    current_start_date = start_date
    while current_start_date <= end_date:
        current_end_date = min(current_start_date + datetime.timedelta(days=batch_days), end_date)
        
        # 3. 获取数据
        f = getattr(pro, api)
        df = f(start_date=current_start_str, end_date=current_end_str)
        
        # 4. 数据预处理
        if df is not None and not df.empty:
            # 去重
            if 'ts_code' in df.columns and 'trade_date' in df.columns:
                df = df.drop_duplicates(subset=['ts_code', 'trade_date'], keep='first')
            
            # 数据类型处理
            for col in df.columns:
                if col in ['ts_code', 'symbol', 'code', 'trade_date']:
                    df.loc[:, col] = df[col].fillna('').astype(str)
            
            # 未来日期检查
            today_str = datetime.datetime.now().strftime('%Y%m%d')
            if 'trade_date' in df.columns:
                df = df[df['trade_date'] <= today_str]
            
            # 5. 分批写入数据库
            chunk_size = 5000
            for i in range(0, len(df), chunk_size):
                chunk_df = df.iloc[i:i+chunk_size].copy()
                DB.safe_to_sql(chunk_df, table, db, if_exists='append')
            
            # 6. 保存检查点
            if 'trade_date' in df.columns and not df.empty:
                max_date = df['trade_date'].max()
                tsAStockPrice.save_checkpoint(api, table, max_date)
        
        current_start_date = current_end_date + datetime.timedelta(days=1)
```

**线程安全表状态管理**：
```python
class TableStateManager:
    """线程安全的表状态管理器"""
    
    def __init__(self):
        self._table_status = {}  # {db_name: {table_name: {'exists': bool, 'creating': bool}}}
        self._lock = threading.RLock()  # 使用可重入锁
    
    def mark_table_creating(self, db, table_name):
        """标记表正在创建中，返回是否成功获取创建权限"""
        with self._lock:
            table_key = table_name
            if table_key in self._table_status[db]:
                current_status = self._table_status[db][table_key]
                if current_status.get('creating', False):
                    return False  # 其他线程已经在创建
                elif current_status.get('exists', False):
                    return False  # 表已存在
            
            # 设置创建中状态
            self._table_status[db][table_key] = {'exists': False, 'creating': True}
            return True  # 成功获取创建权限
    
    def wait_for_table_creation(self, db, table_name, timeout=300):
        """等待表创建完成"""
        start_time = time.time()
        while time.time() - start_time < timeout:
            if not self.is_table_creating(db, table_name):
                return self.is_table_exists(db, table_name)
            time.sleep(1)
        return False
```

### 4. tsAStockIndex - A股指数数据收集器

**主要功能**：
- 获取指数基本信息
- 获取指数日线、周线、月线行情

**核心特性**：
- **多线程并行处理**：将指数列表分成3个部分，每个线程处理一部分
- **多轮次执行**：根据数据延迟情况执行多轮获取
- **线程安全**：使用线程锁保护数据库写入操作

**多线程处理逻辑**：
```python
@staticmethod
def _process_index_daily_worker(pro, db, index_list, today):
    """线程工作函数，处理指数日线数据"""
    table = "astock_index_daily"
    
    for ts_code in index_list:
        try:
            lastdate = tsSHelper.getLastDateAndDelete('astock_index_daily', 'trade_date', ts_code=ts_code, db=db)
            df = pro.index_daily(ts_code=ts_code, start_date=lastdate, end_date=today)
            if not df.empty:
                # 使用线程锁保护数据库写入操作
                with tsAStockIndex._lock:
                    DB.safe_to_sql(df, table, db, index=False, if_exists='append', chunksize=5000)
        except Exception as e:
            Log.logger.error(f"获取指数{ts_code}数据时出错: {str(e)}")

@tsMonitor
def index_daily(pro, db):
    # 多轮次执行逻辑
    n = 3  # 执行3轮
    
    for round_num in range(n):
        # 将指数列表分成3个部分
        index_chunks = [
            index_list[i:i + chunk_size] 
            for i in range(0, len(index_list), chunk_size)
        ]
        
        # 使用线程池执行
        with ThreadPoolExecutor(max_workers=3) as executor:
            futures = []
            for i, chunk in enumerate(index_chunks):
                if chunk:
                    future = executor.submit(
                        tsAStockIndex._process_index_daily_worker, 
                        pro, db, chunk, today
                    )
                    futures.append(future)
            
            # 等待所有线程完成
            for future in futures:
                future.result()
        
        # 轮次间延迟
        if round_num < n - 1:
            time.sleep(10)
```

def check_database_directory(db_name):

        ORDER BY trade_date

```python
def ensure_db_directory(self, db_name):
    """确保数据库目录存在"""
    try:
        db_config = Config.get_config('db', db_name)
        
        if db_config.get('type', '') != 'sqlite':
            return True
        
        db_path = db_config.get('path', '')
        if not db_path:
            Log.logger.warning(f"SQLite数据库配置中未指定path参数")
            return False
        
        # 检查数据库文件是否损坏
        if os.path.exists(db_path):
            try:
                conn = sqlite3.connect(db_path, timeout=10)
                cursor = conn.cursor()
                cursor.execute("SELECT 1")
                conn.close()
            except Exception as e:
                Log.logger.error(f"数据库文件可能已损坏: {str(e)}")
                # 重命名损坏的数据库文件
                backup_path = f"{db_path}.bak.{int(time.time())}"
                os.rename(db_path, backup_path)
                Log.logger.warning(f"已将损坏的数据库文件重命名为: {backup_path}")
        
        return True
    except Exception as e:
        Log.logger.error(f"检查数据库目录时出错: {str(e)}")
        return False
```
