# Finhack因子计算详细流程

## 概述

Finhack项目提供了一个完整的量化金融因子计算框架，支持技术指标因子、Alpha因子、财务因子等多种类型的因子计算。整个系统采用分层架构设计，支持高性能并行计算、缓存优化和依赖关系管理。

## 系统架构

### 核心模块组成

1. **因子计算引擎 (Factor Engines)**
   - `indicatorEngine`: 技术指标因子计算引擎
   - `alphaEngine`: Alpha因子表达式计算引擎

2. **因子管理器 (Factor Manager)**
   - `factorManager`: 统一的因子数据管理和存储
   - 支持矩阵型(matrix)和向量型(vector)因子
   - 时间分片存储和缓存优化

3. **因子分析器 (Factor Analyzer)**
   - `factorAnalyzer`: 因子质量评估和分析
   - IC/IR计算、夏普比率、因子有效性检测

## 因子类型体系

### 1. 技术指标因子 (Technical Indicators)

技术指标因子基于价格、成交量等市场数据计算，包括：

**趋势指标**
- MA系列：SMA, EMA, DEMA, TEMA, TRIMA, WMA
- 布林线：BBANDS (UPPER, MIDDLE, LOWER)
- SAR抛物线：SAR, SAREXT
- 趋势线：HT_TRENDLINE

**动量指标**
- RSI相对强弱指标
- MACD指标：MACD, MACDSIGNAL, MACDHIST
- 随机指标：SLOWK, SLOWD, FASTK, FASTD
- 威廉指标：WILLR
- 动量指标：MOM, ROC, ROCR

**成交量指标**
- OBV累积成交量
- AD累积分布线
- MFI资金流向指标

**波动率指标**
- ATR平均真实波幅
- 标准差：STDDEV, VAR
- 布林线宽度

**财务基本面指标**
- 估值指标：pe, peTtm, pb, ps, psTtm
- 盈利能力：roe, roa, roic
- 成长性：eps增长率、营收增长率等

### 2. Alpha因子 (Alpha Factors)

Alpha因子通过数学表达式定义，支持复杂的逻辑运算：

**表达式语法**
- 基础运算：+, -, *, /, **（幂运算）
- 逻辑运算：&&（与）, ||（或）, !（非）
- 条件运算：三元运算符 condition ? true_value : false_value
- 时间序列函数：delay(), ts_sum(), ts_max(), ts_min()
- 横截面函数：rank(), scale()
- 统计函数：correlation(), covariance(), stddev()

**示例Alpha表达式**
```
# 价格动量因子
close / delay(close, 20) - 1

# 成交量价格关系
rank(volume) * rank(close / delay(close, 1) - 1)

# 条件因子
close > delay(close, 5) ? 1 : -1
```

```python
def where(c, t, f):
    """条件选择：c为True返回t，否则返回f"""
    df = pd.DataFrame({'c': c, 't': t, 'f': f})

def sma(x, n, m=2):
    """平滑移动平均：SMA_t = (x_t * m + SMA_{t-1} * (n-m)) / n"""
    x = x.fillna(0)
    sma_all = []
    grouped = x.groupby('code')
    for name, group in grouped:
        res = group.copy()
        for i in range(1, len(group)):
            res.iloc[i] = (group.iloc[i] * m + group.iloc[i-1] * (n-m)) / float(n)
        res = pd.Series(res, index=group.index)
        sma_all.append(res)
    return pd.concat(sma_all)
```

### 5. 因子质量评估

#### 5.1 IC/IR分析

```python
def ICIR(df, factor_name, period=120):
    """
    信息系数(IC)和信息比率(IR)计算：
    1. 分周期计算因子值与未来收益的相关性
    2. IC = 平均相关系数
    3. IR = IC / IC标准差
    """
    IC_all = []
    df_reset = df.reset_index()
    df_reset['trade_date'] = pd.to_datetime(df_reset['trade_date'], format='%Y%m%d')

    grouped = df_reset.groupby(pd.Grouper(key='trade_date', freq='{}D'.format(period)))

    for name, group in grouped:
        corr = group[factor_name].corr(group['return'])
        if not math.isnan(corr) and not math.isinf(corr):
            IC_all.append(corr)

    if not IC_all:
        return 0, 0

    IC = np.mean(IC_all)
    IC_std = np.std(IC_all, ddof=1)
    IR = IC / IC_std if IC_std != 0 else 0

    return IC, IR

def rank_ICIR(df, factor_name, period=120):
    """
    排名IC/IR计算（Spearman相关系数）

    与ICIR区别：
    - ICIR：Pearson相关系数（因子值与收益）
    - rank_ICIR：Spearman相关系数（排名与收益排名）
    """
    rank_IC_all = []
    df_reset = df.reset_index()
    df_reset['trade_date'] = pd.to_datetime(df_reset['trade_date'], format='%Y%m%d')

    grouped = df_reset.groupby(pd.Grouper(key='trade_date', freq='{}D'.format(period)))

    for name, group in grouped:
        factor_ranks = group[factor_name].rank(method='average')
        return_ranks = group['return'].rank(method='average')
        corr = factor_ranks.corr(return_ranks)
        if not math.isnan(corr) and not math.isinf(corr):
            rank_IC_all.append(corr)

    if not rank_IC_all:
        return 0, 0

    rank_IC = np.mean(rank_IC_all)
    rank_IC_std = np.std(rank_IC_all, ddof=1)
    rank_IR = rank_IC / rank_IC_std if rank_IC_std != 0 else 0

    return rank_IC, rank_IR
```

#### 5.2 夏普比率计算

```python
def Sharpe(df, factor_name, n=10):
    """
    按因子值分组计算加权夏普比率：
    1. 因子值标准化（Z-score）
    2. 标准化后分位数分组
    3. 各组计算夏普比率
    4. 按净值加权平均
    """
    # 1. 标准化
    df['alpha_std'] = df.groupby('trade_date')[factor_name].transform(
        lambda x: (x - x.mean()) / x.std() if x.std() != 0 else 0
    )

    # 2. 分位数分组
    def quantile_cut(group, n):
        try:
            res, bins = pd.qcut(group, q=n, retbins=True, duplicates='drop')
            labels = range(1, len(bins))
            return pd.cut(group, bins=bins, labels=labels, include_lowest=True)
        except ValueError:
            return pd.Series([None] * len(group), index=group.index)

    df['alpha_quantile'] = df.groupby('trade_date')['alpha_std'].transform(lambda x: quantile_cut(x, n))
    df = df.dropna()

    # 3. 计算各分位数组合的收益率和夏普比率
    grouped = df.groupby(['trade_date', 'alpha_quantile'])
    quantile_performance = {}

    for quantile in df['alpha_quantile'].unique():
        mean_df = grouped['return'].mean().fillna(1)
        returns = mean_df.loc[pd.IndexSlice[:, quantile]].tolist()
        excess_returns = [x - 1 for x in returns]

        avg_return = np.mean(excess_returns)
        std_return = np.std(excess_returns)
        sharpe_ratio = avg_return / std_return if std_return != 0 else 0

        quantile_performance[quantile] = {
            'net_value': np.cumprod(returns)[-1],
            'sharpe_ratio': sharpe_ratio
        }

    # 4. 按净值加权平均夏普比率
    total_net_value = sum(item['net_value'] for item in quantile_performance.values())
    weighted_sharpe = sum((item['net_value'] / total_net_value) * item['sharpe_ratio']
                         for item in quantile_performance.values())

    return weighted_sharpe
```

```python
def timeSplit(start_date, end_date, freq):
    """
    根据频率智能分片时间范围：
    - 日/周频率：按年分片
    - 分钟/小时频率：按月分片
    - 秒级频率：按日分片
    """
    start_dt = datetime.datetime.strptime(start_date, "%Y%m%d")
    end_dt = datetime.datetime.strptime(end_date, "%Y%m%d")
    time_ranges = []

    if 'd' in freq or 'w' in freq:
        # 按年分片
        for year in range(start_dt.year, end_dt.year + 1):
            year_start = max(start_dt, datetime.datetime(year, 1, 1))
            year_end = min(end_dt, datetime.datetime(year, 12, 31))
            time_ranges.append((year_start.strftime("%Y%m%d"),
                               year_end.strftime("%Y%m%d")))
    elif 'm' in freq or 'h' in freq:
        # 按月分片
        current = start_dt
        while current <= end_dt:
            month_end = min(end_dt,
                           datetime.datetime(current.year, current.month,
                                           calendar.monthrange(current.year, current.month)[1]))
            time_ranges.append((current.strftime("%Y%m%d"),
                               month_end.strftime("%Y%m%d")))
            if current.month == 12:
                current = datetime.datetime(current.year + 1, 1, 1)
            else:
                current = datetime.datetime(current.year, current.month + 1, 1)

    return time_ranges
```

```python
def getIndicatorList(market, freq, task_list):
    """
    根据任务列表加载指标因子

    参数:
        task_list: 逗号分隔的因子列表名称，如"woldy,all,alpha191"
        market: 市场类型
        freq: 频率

    加载逻辑：
    - 遍历task_list中的每个列表名
    - 读取对应的配置文件
    - 合并所有因子到indicator_list
    """
    indicator_list = []

    # 搜索路径：
    # {CONFIG_DIR}/factorlist/indicatorlist/{market}/{freq}/{list_name}
    # {CONFIG_DIR}/factorlist/indicatorlist/{market}/x{freq}/{list_name}

    for root, dirs, files in os.walk(CONFIG_DIR + f"/factorlist/indicatorlist/{market}/{freq}/"):
        for file in files:
            if file in task_list.split(','):
                with open(os.path.join(root, file), 'r', encoding='utf-8') as f:
                    lines = [line.strip() for line in f.readlines()]
                    processed_lines = []
                    for line in lines:
                        if '_' not in line and line.strip() not in ['open', 'high', 'low', 'close', 'volume']:
                            processed_lines.append(f"{line}_0")
                        else:
                            processed_lines.append(line)
                    indicator_list.extend(processed_lines)
    return indicator_list

def get_alpha_list(market, freq, task_list):
    """
    根据任务列表加载Alpha因子

    Alpha因子配置格式：
    - 每行一个Alpha表达式公式
    - 自动生成因子名称：{文件名}_{行号(3位数字)}
    """
    alpha_list = []

    for root, dirs, files in os.walk(CONFIG_DIR + f"/factorlist/alphalist/{market}/{freq}/"):
        for file in files:
            if file in task_list.split(','):
                with open(os.path.join(root, file), 'r', encoding='utf-8') as f:
                    lines = [line.strip() for line in f.readlines()]
                    file_name = os.path.splitext(file)[0]
                    for i, formula in enumerate(lines, 1):
                        alpha_name = f"{file_name}_{str(i).zfill(3)}"
                        alpha_list.append({"name": alpha_name, "formula": formula})
    return alpha_list
```
